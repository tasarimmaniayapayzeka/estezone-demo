# ==========================================================================
# ESTEZONE DEMO — public_html/.htaccess
# Bu dosyayı sunucuda ".htaccess" adıyla public_html içine koyun
# (baştaki nokta önemli; cPanel Dosya Yöneticisi'nde "Gizli dosyaları göster"
#  seçeneğini açmanız gerekebilir).
# ==========================================================================

# --- HTTPS'e yönlendir ---
RewriteEngine On
RewriteCond %{HTTPS} !=on
RewriteRule ^(.*)$ https://%{HTTP_HOST}/$1 [R=301,L]

# --- .html uzantısız adresler de çalışsın (…/iletisim → iletisim.html) ---
RewriteCond %{REQUEST_FILENAME} !-d
RewriteCond %{REQUEST_FILENAME}\.html -f
RewriteRule ^(.*)$ $1.html [L]

# --- 404 sayfası ---
ErrorDocument 404 /404.html

# --- Kaynak/yardımcı dosyalar dışarıya kapalı ---
<FilesMatch "\.(md|json|log|bak|ornek)$">
  Require all denied
</FilesMatch>

# --- Sıkıştırma ---
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css text/javascript application/javascript image/svg+xml
</IfModule>

# --- Önbellek: varlıklar uzun, HTML kısa (içerik güncellemesi hemen görünsün) ---
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresByType image/webp "access plus 1 year"
  ExpiresByType image/png  "access plus 1 year"
  ExpiresByType video/mp4  "access plus 1 year"
  ExpiresByType text/css   "access plus 1 year"
  ExpiresByType text/javascript "access plus 1 year"
  ExpiresByType text/html  "access plus 10 minutes"
</IfModule>

# --- Güvenlik başlıkları ---
<IfModule mod_headers.c>
  Header always set X-Content-Type-Options "nosniff"
  Header always set X-Frame-Options "SAMEORIGIN"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  # Demo: arama motorları başlık düzeyinde de uzak tutulur
  Header always set X-Robots-Tag "noindex, nofollow"
</IfModule>
